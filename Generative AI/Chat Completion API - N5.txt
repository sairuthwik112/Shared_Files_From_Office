Full Website Summarizer using Ollama


"""
Website summarizer using Ollama instead of OpenAI
"""

from openai import OpenAI
from scraper import fetch_website_contents   # your custom scraper function

# Local Ollama API endpoint
OLLAMA_BASE_URL = "http://localhost:11434/v1"

# Model to use
MODEL = "llama3.2"

# System prompt (defines behavior)
system_prompt = """
You are a snarky assistant that analyzes the contents of a website,
and provides a short, snarky, humorous summary, ignoring text that might be navigation related.
Respond in markdown. Do not wrap the markdown in a code block - respond just with the markdown.
"""

# User prompt (task instruction)
user_prompt_prefix = """
Here are the contents of a website.
Provide a short summary of this website.
If it includes news or announcements, then summarize these too.
"""

# Build messages for LLM
def messages_for(website):
    """Create message list for the LLM."""
    return [
        {"role": "system", "content": system_prompt},
        {"role": "user", "content": user_prompt_prefix + website}
    ]

# Main summarization function
def summarize(url):
    """Fetch and summarize a website using Ollama."""

    # Connect to local Ollama
    ollama = OpenAI(base_url=OLLAMA_BASE_URL, api_key="ollama")

    # Get website content
    website = fetch_website_contents(url)

    # Send to model
    response = ollama.chat.completions.create(
        model=MODEL,
        messages=messages_for(website)
    )

    # Return summary text
    return response.choices[0].message.content

# Main entry point
def main():
    """Main function to run the program."""
    
    url = input("Enter a URL to summarize: ")
    print("\nFetching and summarizing...\n")

    summary = summarize(url)

    print("\n✅ Summary:\n")
    print(summary)

# Run program
if __name__ == "__main__":
    main()
